def is_palindrome(text: str) -> bool: # 대소문자 구분 없이 영문자와 숫자만 추출 cleaned = ''.join(char.lower() for char in text if char.isalnum()) # 뒤집은 문자열과 비교 return cleaned == cleaned[::-1] # 테스트 test_words = ["A man, a plan, a canal: Panama", "Hello World", "토마토"] for word in test_words: result = is_palindrome(word) print(f"'{word}' -> 회문 여부: {result}")
